cmake_minimum_required(VERSION 3.22)
project(dot_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper-build EXCLUDE_FROM_ALL)
add_library(dot_voice SHARED dot_voice.cpp)
target_link_libraries(dot_voice PRIVATE whisper)
if(ANDROID)
    target_link_options(dot_voice PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
    find_package(JNI REQUIRED)
    target_include_directories(dot_voice PRIVATE ${JNI_INCLUDE_DIRS})
endif()
